問題 12: キャッシュ戦略の不整合（Thundering Herd）

📝 シナリオ

大規模ECサイトの商品詳細ページのキャッシュ戦略をレビューしています。キャッシュの有効期限切れ時に大量のリクエストがDBに殺到する「Thundering Herd問題」が発生しています。

以下のコードを読んでください

// === 商品詳細APIのキャッシュ実装 ===
class ProductService {
  private redis: RedisClient;
  private db: DatabaseClient;

  async getProduct(productId: string): Promise<Product> {
    const cacheKey = `product:${productId}`;

    // 1. キャッシュから取得を試みる
    const cached = await this.redis.get(cacheKey);
    if (cached) {
      return JSON.parse(cached);
    }

    // 2. キャッシュミス → DBから取得
    // 問題1: ロックなし → 同時に100リクエストが来ると100回DBクエリ
    const product = await this.db.query(
      'SELECT * FROM products WHERE id = ?', [productId]
    );

    // 3. キャッシュに保存
    await this.redis.set(cacheKey, JSON.stringify(product), {
      EX: 300,  // 全商品一律5分で期限切れ
      // 問題2: 全キーが同じTTL → 大量キーが同時に期限切れ
    });

    return product;
  }

  // 商品更新時のキャッシュ無効化
  async updateProduct(productId: string, data: UpdateData) {
    await this.db.update('products', productId, data);

    // 問題3: DBとキャッシュの更新が非アトミック
    // この間に読み取りが来ると古いデータを返す
    await this.redis.del(`product:${productId}`);

    // 問題4: 関連キャッシュ（カテゴリ一覧、検索結果等）が残る
    // 商品名を変更しても、カテゴリページには古い名前が表示される
  }

  // 全商品キャッシュのウォームアップ
  async warmUpCache() {
    const allProducts = await this.db.query('SELECT * FROM products');
    // 問題5: 全商品を一度にDBから取得 → メモリ溢れの危険
    for (const product of allProducts) {
      await this.redis.set(
        `product:${product.id}`,
        JSON.stringify(product),
        { EX: 300 }  // 問題6: ウォームアップ後も一律300秒で全部同時に切れる
      );
    }
  }
}
