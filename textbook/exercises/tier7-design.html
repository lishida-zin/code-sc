class CouponController {
  async applyCoupon(req: Request, res: Response) {
    try {
      const { cartId, couponCode } = req.body;

      // バリデーション
      if (!cartId || !couponCode) {
        return res.status(400).json({ error: "必須項目が不足しています" });
      }

      // クーポン取得
      const coupon = await db.query(
        "SELECT * FROM coupons WHERE code = $1", [couponCode]
      );
      if (!coupon.rows[0]) {
        return res.status(404).json({ error: "クーポンが見つかりません" });
      }

      const c = coupon.rows[0];

      // 有効期限チェック
      if (new Date(c.expires_at) < new Date()) {
        return res.status(400).json({ error: "クーポンの有効期限切れです" });
      }

      // 使用回数チェック
      if (c.used_count >= c.max_uses) {
        return res.status(400).json({ error: "クーポンの使用上限に達しています" });
      }

      // カート取得
      const cart = await db.query(
        "SELECT * FROM carts WHERE id = $1", [cartId]
      );
      const cartData = cart.rows[0];

      // 最低購入金額チェック
      if (cartData.total < c.min_purchase) {
        return res.status(400).json({
          error: `${c.min_purchase}円以上のお買い物が必要です`
        });
      }

      // 割引計算
      let discount = 0;
      if (c.type === "percentage") {
        discount = Math.floor(cartData.total * (c.value / 100));
        if (c.max_discount) discount = Math.min(discount, c.max_discount);
      } else if (c.type === "fixed") {
        discount = c.value;
      }

      // カート更新
      await db.query(
        "UPDATE carts SET discount = $1, coupon_id = $2 WHERE id = $3",
        [discount, c.id, cartId]
      );

      // クーポン使用回数更新
      await db.query(
        "UPDATE coupons SET used_count = used_count + 1 WHERE id = $1",
        [c.id]
      );

      res.json({ discount, newTotal: cartData.total - discount });
    } catch (err) {
      res.status(500).json({ error: "サーバーエラー" });
    }
  }
